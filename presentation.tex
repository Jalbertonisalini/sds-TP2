  \pgfmathtruncatemacro{\nd}{#2-#1+1}%
  \foreach \fr [count=\k from 0] in {#1,...,#2} {%
    \pgfmathsetmacro{\x}{\bx + \k*\dgap}%
    \ifnum\fr=\cf
      \fill[white] (\x,0) circle(\rad);
      \draw[white,line width=0.8pt] (\x,0) circle(\rad+1.2pt);
    \else\ifnum\fr<\cf
      \fill[white] (\x,0) circle(\rad);
    \else
      \draw[white,line width=0.4pt] (\x,0) circle(\rad);
    \fi\fi
  }%
  \node[white,font=\tiny\bfseries]
    at ({\bx+(\nd-1)*\dgap/2},-0.34) {#3};%
  \pgfmathsetmacro{\bxtmp}{\bx + (\nd-1)*\dgap + \sgap}%
  \let\bx\bxtmp
}

% --- Frame 4 ---
\begin{frame}{Sistema}
  \begin{itemize}
    \setlength{\itemsep}{1em}
    \item Partículas autopropulsadas a rapidez constante en una caja periódica
    \item Interacción local dentro de $r_c$: Vicsek (promedio) o Votante (copia)
    \item El ruido $\eta$ controla la transición entre orden colectivo y desorden
  \end{itemize}
\end{frame}

% --- Frame 5 ---
\begin{frame}{Modelo matemático: vecindad y regla de Vicsek}
  \Large
  \vspace{0.6em}
  \textbf{Conjunto de vecinos}
  \vspace{0.5em}
  \[
    \mathcal{N}_i(t) =
    \left\{j:\left|\mathbf{r}_j(t)-\mathbf{r}_i(t)\right|<r_c\right\}
    \cup \{i\}
  \]
  \vspace{1.5em}
  \textbf{Dirección promedio (Vicsek)}
  \vspace{0.5em}
  \[
    \bar{\theta}_i(t) = \operatorname{atan2}\!\left(
      \sum_{j \in \mathcal{N}_i(t)} \sin\theta_j(t),
      \sum_{j \in \mathcal{N}_i(t)} \cos\theta_j(t)
    \right)
  \]
\end{frame}

% --- Frame 6 ---
\begin{frame}{Modelo matemático: regla de votante y actualización}
  \large
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \textbf{Regla de votante}
    \[
      \bar{\theta}_i(t)=\theta_j(t),
      \qquad j\sim\mathcal{U}\!\left(\mathcal{N}_i(t)\right)
    \]
    \vfill
    \textbf{Dirección}
    \[
      \theta_i(t+1)=\bar{\theta}_i(t)+\eta\,\xi_i(t)
    \]

    \column{0.48\textwidth}
    \textbf{Ruido}
    \[
      \xi_i(t)\sim\mathcal{U}\!\left[-\frac{1}{2},\frac{1}{2}\right]
    \]
    \vfill
    \textbf{Posición}
    \[
      \vv{r}_i(t+1)=\vv{r}_i(t)+v
      \begin{pmatrix}
        \cos\theta_i(t+1)\\
        \sin\theta_i(t+1)
      \end{pmatrix}
    \]
  \end{columns}
\end{frame}

% --- Frame 12 ---
\begin{frame}{Observables: polarización}
  \vfill
  \Large
  \textbf{Polarización}
  \[
    v_a(t) = \left|
      \frac{1}{N} \sum_{i=1}^{N}
      \begin{pmatrix}
        \cos\theta_i(t) \\
        \sin\theta_i(t)
      \end{pmatrix}
    \right|
  \]
  \vfill
  \textbf{Promedio estacionario}
  \[
    \langle v_a\rangle =
    \frac{1}{t_{\mathrm{fin}}-t_{\mathrm{ini}}+1}
    \sum_{t=t_{\mathrm{ini}}}^{t_{\mathrm{fin}}} v_a(t)
  \]
  \vfill
\end{frame}

% --- Frame 13 ---
\begin{frame}{Observables: cluster más grande}
  \vfill
  \Large
  \textbf{Fracción del cluster más grande}
  \[
    S(t)=\frac{N_{\mathrm{cluster}}(t)}{N}
  \]
  \vfill
  \textbf{Promedio estacionario}
  \[
    \langle S\rangle =
    \frac{1}{t_{\mathrm{fin}}-t_{\mathrm{ini}}+1}
    \sum_{t=t_{\mathrm{ini}}}^{t_{\mathrm{fin}}} S(t)
  \]
  \vfill
\end{frame}
